cmake_minimum_required(VERSION 3.20)
project(BruteForceRecomp LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/private_branding.cmake)
option(BF_GRAPHICS_ONLY "Build the standalone DX12/FSR GPU verification executable without game data" OFF)
if(BF_GRAPHICS_ONLY)
    include(cmake/graphics.cmake)
    bf_add_dx12_library()
    add_executable(bf_graphics_smoke tests/graphics_smoke.cpp)
    target_link_libraries(bf_graphics_smoke PRIVATE bf_dx12)
    return()
endif()
option(BF_LAUNCHER_ONLY "Build the native disc-import application independently of generated game code" OFF)
if(BF_LAUNCHER_ONLY)
    add_subdirectory(native_launcher)
    return()
endif()
if(MSVC)
    # Upstream runtime targets explicitly use /O2, which conflicts with the
    # default MSVC Debug /RTC1 instrumentation. Keep symbols and /Od for our
    # generated code while allowing those runtime targets to compile.
    string(REPLACE "/RTC1" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
endif()
set(XBOXRECOMP_DIR "" CACHE PATH "Pinned xboxrecomp checkout")
set(BF_GENERATED_DIR "" CACHE PATH "Private generated C source directory")
set(BF_RECOVERED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.local/recovered" CACHE PATH "Private recovered callback source directory")
if(ANDROID)
    message(FATAL_ERROR "Android is not integrated yet. See platform-references.json for the public Vulkan/SDL2 runtime candidate and README.md for the app-lifecycle and device validation gates.")
endif()
if(NOT EXISTS "${XBOXRECOMP_DIR}/CMakeLists.txt" OR NOT EXISTS "${BF_GENERATED_DIR}/recomp_dispatch.c")
    message(FATAL_ERROR "Run scripts/pipeline.py generate first, then set XBOXRECOMP_DIR and BF_GENERATED_DIR.")
endif()
if(NOT EXISTS "${BF_GENERATED_DIR}/native-media-profile.json")
    message(FATAL_ERROR "Prepare the user-approved native media profile with scripts/prepare_native_media.py before building.")
endif()
add_subdirectory("${XBOXRECOMP_DIR}" "${CMAKE_BINARY_DIR}/xboxrecomp")
include(cmake/runtime_fixes.cmake)
bf_apply_runtime_fixes("${XBOXRECOMP_DIR}")
include(cmake/render_profile.cmake)
include(cmake/posix_runtime.cmake)
include(cmake/memory_statistics.cmake)
include(cmake/pushbuffer_stream.cmake)
include(cmake/frame_handoff.cmake)
include(cmake/graphics.cmake)
enable_testing()
add_executable(bf_import_guard_test tests/import_guard_test.cpp)
target_include_directories(bf_import_guard_test PRIVATE src)
add_test(NAME import_guard COMMAND bf_import_guard_test)
add_executable(bf_memory_stats_test tests/guest_memory_stats_test.cpp)
target_include_directories(bf_memory_stats_test PRIVATE src)
add_test(NAME guest_memory_stats COMMAND bf_memory_stats_test)
add_executable(bf_render_blend_test tests/render_blend_test.cpp)
target_include_directories(bf_render_blend_test PRIVATE src)
add_test(NAME opaque_blending COMMAND bf_render_blend_test)
add_executable(bf_pushbuffer_test tests/pushbuffer_test.cpp "${BF_PB_SCANNER}")
target_include_directories(bf_pushbuffer_test PRIVATE src)
add_test(NAME pushbuffer_stream COMMAND bf_pushbuffer_test)
set_tests_properties(pushbuffer_stream PROPERTIES ENVIRONMENT "RECOMP_PB_EXEC=1")
add_executable(bf_gpu_notification_test tests/gpu_notification_test.cpp src/gpu_completion.cpp src/gpu_pushbuffer_fixups.cpp)
target_include_directories(bf_gpu_notification_test PRIVATE "${XBOXRECOMP_DIR}/src/kernel" "${XBOXRECOMP_DIR}/src")
add_test(NAME gpu_fence_notification COMMAND bf_gpu_notification_test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    add_executable(bf_posix_mmio_test tests/posix_mmio_test.c)
    add_test(NAME posix_mmio_context COMMAND bf_posix_mmio_test)
endif()
file(GLOB BF_GENERATED CONFIGURE_DEPENDS "${BF_GENERATED_DIR}/*.c")
include(cmake/recovered_entries.cmake)
include(cmake/bink_trace.cmake)
include(cmake/menu_trace.cmake)
include(cmake/game_text_input.cmake)
include(cmake/game_menu_input.cmake)
include(cmake/physics_trace.cmake)
add_executable(bruteforce_recomp src/main.cpp src/boot_trace.cpp src/file_hash.cpp
    src/native_media.cpp src/instruction_services.cpp src/startup_sampler.cpp
    src/gpu_completion.cpp src/gpu_pushbuffer_fixups.cpp src/host_input.cpp src/host_audio.cpp src/guest_failure.cpp "${BF_RECOVERED_DIR}/recomp_recovered.c" ${BF_GENERATED})
set(BF_HOST_OUTPUT_DIR "" CACHE PATH "Optional separate output directory for a test host; preserves an open player build")
if(BF_HOST_OUTPUT_DIR)
    set_target_properties(bruteforce_recomp PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${BF_HOST_OUTPUT_DIR}")
endif()
target_include_directories(bruteforce_recomp PRIVATE
    "${BF_GENERATED_DIR}" "${BF_GENERATED_DIR}/.."
    "${XBOXRECOMP_DIR}/templates/runtime" "${XBOXRECOMP_DIR}/include")
target_link_libraries(bruteforce_recomp PRIVATE xboxrecomp)
if(WIN32)
    add_custom_command(TARGET bruteforce_recomp POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:bruteforce_recomp>/licenses"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PROJECT_SOURCE_DIR}/third_party/fsr1/LICENSE.txt"
            "$<TARGET_FILE_DIR:bruteforce_recomp>/licenses/AMD-FSR1-LICENSE.txt")
endif()
bf_attach_private_icon(bruteforce_recomp)
if(WIN32)
    target_link_libraries(bruteforce_recomp PRIVATE dbghelp)
endif()
if(MSVC)
    target_compile_options(bruteforce_recomp PRIVATE /bigobj /fp:strict)
    target_link_options(bruteforce_recomp PRIVATE "/MAP:${CMAKE_BINARY_DIR}/bruteforce_recomp.map")
else()
    target_compile_options(bruteforce_recomp PRIVATE -fno-fast-math -ffp-contract=off)
endif()
